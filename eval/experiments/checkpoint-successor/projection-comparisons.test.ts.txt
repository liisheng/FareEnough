const factData = (review: ReturnType<typeof reviewFixtureFromRaw>, id: string) => {
  const fact = review.snapshot.evidence.find(item => item.id === id);

describe('mechanical policy comparisons exposed as evidence', () => {
  it('surfaces no-show numerical conditions separately from location and the final decision', () => {
    const review = loadReviewFixture('DISP-002');
    expect(factData(review, 'derived-ryde-v21-wait')).toMatchObject({ policyEligibleWaitSeconds: 360, requiredMinimumWaitSeconds: 300, meetsMinimumWaitDuration: true, policyRuleIds: ['RYDE-V2.1-2.1'] });
    expect(factData(review, 'derived-ryde-v21-wait').observationLimit).toContain('Sampled locations are not continuous measurement');
    expect(factData(review, 'derived-ryde-v21-contact')).toMatchObject({ driverChatAttemptsBeforeCancellation: 4, driverInAppCallAttemptsBeforeCancellation: 1, requiredMinimumChatAttempts: 1, requiredMinimumInAppCallAttempts: 1, meetsChatAttemptMinimum: true, meetsInAppCallAttemptMinimum: true, policyRuleIds: ['RYDE-V2.1-2.3'] });
    expect(factData(review, 'derived-ryde-v21-fee')).toMatchObject({ recordedFeeCents: 500, requiredStandardFeeCents: 500, matchesStandardFee: true, policyRuleIds: ['RYDE-V2.1-2.2'] });
    expect(factData(review, 'derived-ryde-v21-cancellation')).toMatchObject({ cancellationActor: 'driver', sourceEvidenceIds: ['chat-006'], policyRuleIds: ['RYDE-V2.1-2.2'] });
  });
  it.each([['2026-09-13T08:49:59+08:00', 299, false], ['2026-09-13T08:50:00+08:00', 300, true]])('compares the exact minimum at %s without changing other conditions', (cancellationTime, seconds, meetsMinimumWaitDuration) => {
    const raw = rawNoShow();
    raw.trip_data.cancellation_time = cancellationTime;
    expect(factData(reviewFixtureFromRaw(raw), 'derived-ryde-v21-wait')).toMatchObject({ policyEligibleWaitSeconds: seconds, meetsMinimumWaitDuration });
  });
  it.each([null, 'invalid timestamp'])('keeps a missing or invalid scheduled wait unknown: %s', scheduledTime => {
    const raw = rawNoShow();
    raw.trip_data.scheduled_time = scheduledTime;
    expect(factData(reviewFixtureFromRaw(raw), 'derived-ryde-v21-wait')).toMatchObject({ policyEligibleWaitSeconds: null, meetsMinimumWaitDuration: null });
  });
  it('distinguishes zero recorded attempts from unavailable contact logs', () => {
    const empty = rawNoShow();
    empty.chat_logs = [];
    empty.app_events = [];
    expect(factData(reviewFixtureFromRaw(empty), 'derived-ryde-v21-contact')).toMatchObject({ driverChatAttemptsBeforeCancellation: 0, driverInAppCallAttemptsBeforeCancellation: 0, meetsChatAttemptMinimum: false, meetsInAppCallAttemptMinimum: false });
    const missing = rawNoShow();
    delete missing.chat_logs;
    delete missing.app_events;
    expect(factData(reviewFixtureFromRaw(missing), 'derived-ryde-v21-contact')).toMatchObject({ driverChatAttemptsBeforeCancellation: null, driverInAppCallAttemptsBeforeCancellation: null, meetsChatAttemptMinimum: null, meetsInAppCallAttemptMinimum: null });
  });
  it('does not turn contact records without a cancellation timestamp into successful minimum checks', () => {
    const raw = rawNoShow();
    raw.trip_data.cancellation_time = null;
    expect(factData(reviewFixtureFromRaw(raw), 'derived-ryde-v21-contact')).toMatchObject({ meetsChatAttemptMinimum: null, meetsInAppCallAttemptMinimum: null });
  });
  it('uses the actual actor-identifying record ID rather than a fixture-specific position', () => {
    const raw = rawNoShow();
    raw.chat_logs.unshift({ timestamp: '2026-09-13T08:30:00+08:00', sender: 'rider', type: 'message', content: 'Booking confirmed.' });
    const review = reviewFixtureFromRaw(raw);
    expect(factData(review, 'derived-ryde-v21-cancellation')).toMatchObject({ cancellationActor: 'driver', sourceEvidenceIds: ['chat-007'] });
    expect(review.snapshot.evidence.find(item => item.id === 'chat-007')?.data.content).toContain('Trip cancelled by driver');
  });
  it('does not infer the cancellation actor from timestamps, a fee or rider-no-show reason alone', () => {
    const raw = rawNoShow();
    raw.chat_logs[5].content = 'Trip cancelled. Reason: rider_no_show. Cancellation fee of $5.00 applied.';
    expect(factData(reviewFixtureFromRaw(raw), 'derived-ryde-v21-cancellation')).toMatchObject({ cancellationActor: null, sourceEvidenceIds: [] });
  });
  it.each([[4.99, false], [5, true], [5.01, false], [null, null]])('compares fee %s against the fixed standard without manufacturing an amount', (fee, matchesStandardFee) => {

    const policyIds = new Set(review.snapshot.policyRules.map(item => item.id));
    const comparisons = review.snapshot.evidence.filter(item => item.kind === 'derived_fact' && Array.isArray(item.data.policyRuleIds));
